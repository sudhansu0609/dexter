236	    def _provider_chain(self, provider: str, needs_tools: bool = False) -> List[str]:
237	        """
238	        Ordered providers to try for one request. An explicit choice is honoured
239	        exactly; "auto" is local-first with cloud fallback so a local server
240	        dying mid-request degrades to a configured key instead of an apology.
241	        Gemini is chat-only here (its path has no tool calling), so it is
242	        skipped whenever tools are required.
243	        """
244	        if provider and provider != "auto":
245	            return [provider]
246	        health = self.get_available_providers()
247	        chain: List[str] = []
248	        if health.get("llamacpp", {}).get("online"):
249	            chain.append("llamacpp")
250	        if health["lmStudio"]["online"]:
251	            chain.append("lmstudio")
252	        if health["ollama"]["online"]:
253	            chain.append("ollama")
254	        if health["openai"]["configured"]:
255	            chain.append("openai")
256	        if health["gemini"]["configured"] and not needs_tools:
257	            chain.append("gemini")
258	        # Honour the configured default: move it to the front so it is tried
259	        # first, while the others remain as graceful fallbacks. Frees the
260	        # non-preferred local backend (e.g. LM Studio) for other work.
261	        if self.default_provider != "auto" and self.default_provider in chain:
262	            chain = [self.default_provider] + [p for p in chain if p != self.default_provider]
263	        return chain or ["lmstudio"]

267	    def chat(
268	        self,
269	        messages: List[Dict[str, Any]],
270	        provider: str = "auto",
271	        model: Optional[str] = None,
272	        temperature: float = 0.5,
273	        max_tokens: int = 768,
274	        tools: Optional[List[Dict[str, Any]]] = None,
275	        timeout: int = 90,
276	        on_token: Optional[Callable[[str], None]] = None,
277	        on_fallback: Optional[Callable[[str, str], None]] = None,
278	        should_cancel: Optional[Callable[[], bool]] = None,
279	    ) -> LLMResponse:
280	        """
281	        Full multi-turn chat with optional native tool calling.
282	
283	        `on_token` streams content deltas as they arrive (the request is made
284	        with stream=True); the returned LLMResponse still carries the complete
285	        content and parsed tool calls either way.
286	
287	        `on_fallback(failed, next)` fires before each retry against the next
288	        provider in the chain, so callers can reset any tokens already shown.
289	
290	        `should_cancel()` is polled while a stream is consumed; when it returns
291	        True generation stops and the response is flagged `cancelled` - returned
292	        as-is without trying the next provider (a barge-in is not a failure).
293	        """
294	        chain = self._provider_chain(provider, needs_tools=bool(tools))
295	        last: Optional[LLMResponse] = None
296	        for idx, prov in enumerate(chain):
297	            if should_cancel and should_cancel():
298	                return LLMResponse(provider=prov, model=model or "", cancelled=True)
299	            if idx > 0:
300	                print(f"[HybridLLM] {chain[idx - 1]} failed mid-request; trying {prov}.")
301	                if on_fallback:
302	                    try:
303	                        on_fallback(chain[idx - 1], prov)
304	                    except Exception:
305	                        pass
306	            response = self._chat_one(
307	                prov, messages, model, temperature, max_tokens, tools, timeout,
308	                on_token, should_cancel,
309	            )
310	            if response.cancelled:
311	                return response
312	            if response.ok:
313	                if idx > 0:
314	                    response.fallback_from = chain[0]
315	                return response
316	            last = response
317	            # The failed probe result is stale now; let the next request re-probe.
318	            self.invalidate_health()
319	        return last or LLMResponse(
320	            provider=chain[0] if chain else "none",
321	            error="No LLM provider is available.",
322	        )
